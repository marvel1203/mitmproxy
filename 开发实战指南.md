### 2.1 场景一：修改 HTTP 请求头

**需求**：为所有请求添加自定义头部，记录请求时间。

**实现步骤**：

1. 创建插件文件 `addons/timestamp_header.py`

```python
"""
为所有 HTTP 请求添加时间戳头部
"""
import time
from mitmproxy import http


class TimestampHeader:
    """添加时间戳头部的插件"""
    
    def request(self, flow: http.HTTPFlow) -> None:
        """
        在请求发送到服务器前添加时间戳头部
        
        参数:
            flow: HTTP 流量对象
        """
        # 添加当前时间戳
        timestamp = str(int(time.time() * 1000))
        flow.request.headers["X-Request-Timestamp"] = timestamp
        
        # 记录到日志
        self.log(f"添加时间戳头部: {timestamp} for {flow.request.url}")
    
    def log(self, message: str) -> None:
        """输出日志"""
        print(f"[TimestampHeader] {message}")


# 注册插件
addons = [TimestampHeader()]
```

2. 使用插件

```bash
# 使用 mitmdump
mitmdump -s addons/timestamp_header.py

# 使用 mitmproxy
mitmproxy -s addons/timestamp_header.py
```

3. 添加配置选项

```python
from mitmproxy import http, ctx


class TimestampHeader:
    def load(self, loader):
        """加载配置选项"""
        loader.add_option(
            name="timestamp_header_name",
            typespec=str,
            default="X-Request-Timestamp",
            help="时间戳头部名称"
        )
    
    def request(self, flow: http.HTTPFlow) -> None:
        timestamp = str(int(time.time() * 1000))
        # 使用配置的头部名称
        header_name = ctx.options.timestamp_header_name
        flow.request.headers[header_name] = timestamp
```

使用自定义选项：

### 2.2 场景二：拦截并修改响应内容

**需求**：将所有 HTML 页面中的特定文本替换为其他文本。

**实现**：

```python
"""
替换 HTML 响应中的文本内容
"""
import re
from mitmproxy import http


class ContentReplacer:
    """内容替换插件"""
    
    def __init__(self):
        self.replacements = {
            "旧文本": "新文本",
            "example": "示例",
        }
    
    def response(self, flow: http.HTTPFlow) -> None:
        """
        修改响应内容
        
        仅处理 HTML 响应
        """
        # 检查是否是 HTML 内容
        if not self.is_html(flow):
            return
        
        # 获取响应内容
        content = flow.response.get_text()
        
        # 执行替换
        modified = False
        for old_text, new_text in self.replacements.items():
            if old_text in content:
                content = content.replace(old_text, new_text)
                modified = True
        
        # 如果有修改，更新响应
        if modified:
            flow.response.set_text(content)
            print(f"[ContentReplacer] 修改了 {flow.request.url}")
    
    def is_html(self, flow: http.HTTPFlow) -> bool:
        """检查是否是 HTML 响应"""
        if not flow.response:
            return False
        
        content_type = flow.response.headers.get("content-type", "")
        return "text/html" in content_type.lower()


addons = [ContentReplacer()]
```

**高级版本：支持正则表达式**

```python
class AdvancedContentReplacer:
    """高级内容替换（支持正则）"""
    
    def __init__(self):
        # 编译正则表达式模式
        self.patterns = [
            (re.compile(r'(\d{3})-(\d{4})-(\d{4})'), r'\1****\3'),  # 隐藏手机号中间4位
            (re.compile(r'password\s*=\s*"[^"]*"'), 'password="***"'),  # 隐藏密码
        ]
    
    def response(self, flow: http.HTTPFlow) -> None:
        if not self.is_html(flow):
            return
        
        content = flow.response.get_text()
        
        # 应用所有正则替换
        for pattern, replacement in self.patterns:
            content = pattern.sub(replacement, content)
        
        flow.response.set_text(content)
```

### 2.3 场景三：实现请求缓存

**需求**：缓存 GET 请求的响应，相同请求直接返回缓存。

**实现**：

```python
"""
简单的 HTTP 缓存实现
"""
import hashlib
import json
import time
from typing import Dict, Tuple
from mitmproxy import http


class SimpleCache:
    """简单的 HTTP 缓存"""
    
    def __init__(self):
        # 缓存存储: cache_key → (response, timestamp)
        self.cache: Dict[str, Tuple[http.Response, float]] = {}
        self.cache_ttl = 300  # 缓存有效期（秒）
        self.hit_count = 0
        self.miss_count = 0
    
    def request(self, flow: http.HTTPFlow) -> None:
        """
        检查缓存，如果有则直接返回
        """
        # 只缓存 GET 请求
        if flow.request.method != "GET":
            return
        
        # 生成缓存键
        cache_key = self.get_cache_key(flow.request)
        
        # 检查缓存
        if cache_key in self.cache:
            cached_response, timestamp = self.cache[cache_key]
            
            # 检查是否过期
            if time.time() - timestamp < self.cache_ttl:
                # 缓存命中，直接返回缓存的响应
                flow.response = cached_response.copy()
                flow.response.headers["X-Cache"] = "HIT"
                self.hit_count += 1
                print(f"[Cache] HIT: {flow.request.url}")
                return
            else:
                # 缓存过期，删除
                del self.cache[cache_key]
        
        # 缓存未命中
        self.miss_count += 1
        print(f"[Cache] MISS: {flow.request.url}")
    
    def response(self, flow: http.HTTPFlow) -> None:
        """
        缓存响应
        """
        # 只缓存 GET 请求的成功响应
        if flow.request.method != "GET":
            return
        
        if not flow.response or flow.response.status_code != 200:
            return
        
        # 检查是否应该缓存（根据 Cache-Control 头）
        if not self.should_cache(flow.response):
            return
        
        # 存储到缓存
        cache_key = self.get_cache_key(flow.request)
        self.cache[cache_key] = (flow.response.copy(), time.time())
        flow.response.headers["X-Cache"] = "MISS"
        
        print(f"[Cache] 已缓存: {flow.request.url} (总数: {len(self.cache)})")
    
    def get_cache_key(self, request: http.Request) -> str:
        """生成缓存键"""
        # 使用 URL 和查询参数生成唯一键
        key_data = f"{request.method}:{request.url}"
        return hashlib.md5(key_data.encode()).hexdigest()
    
    def should_cache(self, response: http.Response) -> bool:
        """判断响应是否应该缓存"""
        cache_control = response.headers.get("cache-control", "").lower()
        
        # 不缓存明确声明的内容
        if "no-store" in cache_control or "no-cache" in cache_control:
            return False
        
        return True
    
    def done(self):
        """插件卸载时输出统计"""
        total = self.hit_count + self.miss_count
        if total > 0:
            hit_rate = (self.hit_count / total) * 100
            print(f"\n[Cache] 统计:")
            print(f"  总请求: {total}")
            print(f"  命中: {self.hit_count}")
            print(f"  未命中: {self.miss_count}")
            print(f"  命中率: {hit_rate:.2f}%")
            print(f"  缓存条目: {len(self.cache)}")


addons = [SimpleCache()]
```

### 2.4 场景四：实现流量录制与重放

**需求**：录制特定网站的流量，并能够重放这些请求。

**实现**：

```python
"""
流量录制与重放
"""
import json
import pickle
from pathlib import Path
from mitmproxy import http, flow as mitmflow


class FlowRecorder:
    """流量录制器"""
    
    def __init__(self, output_file: str = "recorded_flows.json"):
        self.output_file = Path(output_file)
        self.flows = []
        self.recording = True
    
    def response(self, flow: http.HTTPFlow) -> None:
        """记录完整的流量"""
        if not self.recording:
            return
        
        # 只记录成功的响应
        if not flow.response or flow.response.status_code >= 400:
            return
        
        # 序列化流量信息
        flow_data = {
            "url": flow.request.url,
            "method": flow.request.method,
            "headers": dict(flow.request.headers),
            "status_code": flow.response.status_code,
            "response_headers": dict(flow.response.headers),
            "response_body": flow.response.content.decode("utf-8", errors="ignore")
        }
        
        self.flows.append(flow_data)
        print(f"[Recorder] 已记录: {flow.request.method} {flow.request.url}")
    
    def done(self):
        """保存录制的流量"""
        if not self.flows:
            print("[Recorder] 没有录制任何流量")
            return
        
        with open(self.output_file, "w", encoding="utf-8") as f:
            json.dump(self.flows, f, indent=2, ensure_ascii=False)
        
        print(f"[Recorder] 已保存 {len(self.flows)} 个流量到 {self.output_file}")


class FlowPlayer:
    """流量重放器"""
    
    def __init__(self, input_file: str = "recorded_flows.json"):
        self.input_file = Path(input_file)
        self.flows = []
        self.current_index = 0
    
    def load(self, loader):
        """加载录制的流量"""
        if not self.input_file.exists():
            print(f"[Player] 文件不存在: {self.input_file}")
            return
        
        with open(self.input_file, "r", encoding="utf-8") as f:
            self.flows = json.load(f)
        
        print(f"[Player] 已加载 {len(self.flows)} 个流量")
    
    def request(self, flow: http.HTTPFlow) -> None:
        """使用录制的响应替代实际请求"""
        if not self.flows or self.current_index >= len(self.flows):
            return
        
        # 查找匹配的流量
        recorded_flow = None
        for recorded in self.flows:
            if (recorded["method"] == flow.request.method and 
                recorded["url"] == flow.request.url):
                recorded_flow = recorded
                break
        
        if not recorded_flow:
            return
        
        # 创建响应（不发送实际请求）
        flow.response = http.Response.make(
            status_code=recorded_flow["status_code"],
            content=recorded_flow["response_body"].encode("utf-8"),
            headers=recorded_flow["response_headers"]
        )
        
        flow.response.headers["X-Replay"] = "true"
        print(f"[Player] 重放: {flow.request.method} {flow.request.url}")


# 使用录制器
addons = [FlowRecorder("my_flows.json")]

# 或使用重放器
# addons = [FlowPlayer("my_flows.json")]
```

```python
"""
WebSocket 消息处理
"""
import json
from mitmproxy import http


class WebSocketFilter:
    """WebSocket 消息过滤器"""
    
    def websocket_message(self, flow: http.HTTPFlow):
        """
        处理 WebSocket 消息
        
        flow.websocket.messages 是消息列表
        最后一条消息是刚接收到的
        """
        message = flow.websocket.messages[-1]
        
        # 判断消息方向
        direction = "客户端→服务器" if message.from_client else "服务器→客户端"
        
        # 获取消息内容
        content = message.content
        
        # 如果是文本消息，尝试解析 JSON
        if message.type == 1:  # TEXT
            try:
                data = json.loads(content)
                print(f"[WebSocket] {direction}: {data}")
                
                # 修改消息内容
                if message.from_client and "password" in data:
                    data["password"] = "***"
                    message.content = json.dumps(data)
                    print(f"[WebSocket] 已隐藏密码字段")
                
            except json.JSONDecodeError:
                print(f"[WebSocket] {direction}: {content[:100]}")
        
        # 如果是二进制消息
        elif message.type == 2:  # BINARY
            print(f"[WebSocket] {direction}: 二进制数据 ({len(content)} 字节)")
    
    def websocket_start(self, flow: http.HTTPFlow):
        """WebSocket 连接建立"""
        print(f"[WebSocket] 连接建立: {flow.request.url}")
    
    def websocket_end(self, flow: http.HTTPFlow):
        """WebSocket 连接关闭"""
        print(f"[WebSocket] 连接关闭: {flow.request.url}")
        print(f"  总消息数: {len(flow.websocket.messages)}")


addons = [WebSocketFilter()]
```

### 3.2 添加新的代理模式

**场景**：实现一个智能路由代理模式，根据域名选择不同的上游代理。

**文件**：`mitmproxy/proxy/mode_specs.py`

```python
@dataclass
class SmartRouteMode:
    """
    智能路由模式
    
    根据目标域名选择不同的上游代理
    """
    routes: dict[str, tuple[str, int]]  # 域名 → (上游代理地址, 端口)
    default_upstream: tuple[str, int] | None = None
    
    def get_upstream(self, host: str) -> tuple[str, int] | None:
        """获取指定主机的上游代理"""
        # 精确匹配
        if host in self.routes:
            return self.routes[host]
        
        # 域名后缀匹配
        for pattern, upstream in self.routes.items():
            if pattern.startswith("*.") and host.endswith(pattern[2:]):
                return upstream
        
        return self.default_upstream
```

**使用新模式**：

```python
# 在 ConnectionHandler 中使用
def make_top_layer(self, context: Context) -> Layer:
    if isinstance(context.client.proxy_mode, SmartRouteMode):
        # 获取目标主机

        if upstream:
            # 创建上游代理层
            return UpstreamProxyLayer(context, upstream)
    
    # 默认逻辑
```

### 3.3 扩展命令系统

**场景**：添加一个命令来统计特定域名的流量。

**文件**：`mitmproxy/addons/trafficstats.py`

```python
from mitmproxy import command, ctx, http
from typing import Sequence


class TrafficStats:
    """流量统计插件"""
    
    def __init__(self):
        self.stats = {}  # 域名 → (请求数, 总字节数)
    
    def response(self, flow: http.HTTPFlow):
        """统计响应"""
        host = flow.request.host
        
        if host not in self.stats:
            self.stats[host] = {"count": 0, "bytes": 0}
        
        self.stats[host]["count"] += 1
        if flow.response:
            self.stats[host]["bytes"] += len(flow.response.content or b"")
    
    @command.command("stats.show")
    def show_stats(self) -> str:
        """
        显示流量统计
        
        命令: stats.show
        """
        if not self.stats:
            return "没有流量统计数据"
        
        lines = ["流量统计:", ""]
        for host, data in sorted(self.stats.items(), key=lambda x: x[1]["bytes"], reverse=True):
            lines.append(f"{host}:")
            lines.append(f"  请求数: {data['count']}")
            lines.append(f"  总字节: {data['bytes']:,} ({self.format_bytes(data['bytes'])})")
            lines.append("")
        
        return "\n".join(lines)
    
    @command.command("stats.reset")
    def reset_stats(self) -> str:
        """
        重置统计
        
        命令: stats.reset
        """
        count = len(self.stats)
        self.stats.clear()
        return f"已重置 {count} 个主机的统计数据"
    
    @command.command("stats.export")
    @command.argument("path", type=command.Path("w"))
    def export_stats(self, path: str) -> str:
        """
        导出统计到文件
        
        命令: stats.export /path/to/file.json
        """
        import json
        with open(path, "w") as f:
            json.dump(self.stats, f, indent=2)
        return f"已导出统计到 {path}"
    
    @staticmethod
    def format_bytes(bytes: int) -> str:
        """格式化字节数"""
        for unit in ["B", "KB", "MB", "GB"]:
            if bytes < 1024:
                return f"{bytes:.2f} {unit}"
            bytes /= 1024
        return f"{bytes:.2f} TB"


addons = [TrafficStats()]
```

## 4. 性能优化

### 4.1 使用异步处理避免阻塞

**问题**：插件中的耗时操作会阻塞其他请求。

**解决方案**：使用异步处理。

```python
import asyncio
import aiohttp
from mitmproxy import http


class AsyncAddon:
    """异步插件示例"""
    
    async def request(self, flow: http.HTTPFlow):
        """
        异步请求处理
        
        使用 async/await 不会阻塞其他请求
        """
        # 异步 HTTP 请求（如调用外部 API）
        async with aiohttp.ClientSession() as session:
            async with session.get(
                "https://api.example.com/check",
                params={"url": flow.request.url}
            ) as resp:
                if resp.status == 403:
                    # 阻止请求
                    flow.response = http.Response.make(403, b"Blocked by filter")
        
        # 异步数据库查询
        user_id = flow.request.headers.get("X-User-ID")
        if user_id:
            user_data = await self.database.get_user(user_id)
            flow.request.headers["X-User-Name"] = user_data["name"]
    
    async def response(self, flow: http.HTTPFlow):
        """异步响应处理"""
        # 异步保存到数据库
        await self.database.save_flow({
            "url": flow.request.url,
            "status": flow.response.status_code,
            "timestamp": time.time()
        })
```

### 4.2 使用线程池处理 CPU 密集型任务

```python
from mitmproxy import http
from mitmproxy.utils import asyncio_utils
import hashlib


class ImageProcessor:
    """图片处理插件"""
    
    @asyncio_utils.concurrent  # 在线程池中执行
    def response(self, flow: http.HTTPFlow):
        """
        处理图片响应
        
        @concurrent 装饰器将方法在线程池中执行，
        不会阻塞主事件循环
        """
        if not self.is_image(flow):
            return
        
        # CPU 密集型操作
        content = flow.response.content
        
        # 计算哈希
        hash_value = hashlib.sha256(content).hexdigest()
        flow.response.headers["X-Image-Hash"] = hash_value
        
        # 图片处理（如添加水印）
        processed = self.add_watermark(content)
        flow.response.content = processed
    
    def is_image(self, flow: http.HTTPFlow) -> bool:
        """检查是否是图片"""
        if not flow.response:
            return False
        content_type = flow.response.headers.get("content-type", "")
        return content_type.startswith("image/")
    
    def add_watermark(self, image_data: bytes) -> bytes:
        """添加水印（示例）"""
        # 实际实现需要使用 PIL/Pillow
        return image_data  # 简化示例
```

### 4.3 流式处理大文件

```python
from mitmproxy import http


class StreamingHandler:
    """流式处理大文件"""
    
    def responseheaders(self, flow: http.HTTPFlow):
        """
        在接收响应头时判断是否需要流式处理
        
        这个钩子在响应体到达前触发
        """
        # 检查 Content-Length
        content_length = flow.response.headers.get("content-length", "0")
        size = int(content_length)
        
        # 大于 10MB 的响应使用流式处理
        if size > 10 * 1024 * 1024:
            # 启用流式处理
            flow.response.stream = True
            print(f"[Streaming] 启用流式处理: {flow.request.url} ({size:,} 字节)")
    
    def response(self, flow: http.HTTPFlow):
        """
        处理完整响应
        
        如果启用了流式处理，这个钩子可能不会被调用，
        或者 flow.response.content 是 None
        """
        if flow.response.stream:
            print(f"[Streaming] 流式传输完成: {flow.request.url}")
```

```python
import logging

logger = logging.getLogger(__name__)
```
